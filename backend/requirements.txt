# --- Framework y Servidor ---
fastapi                     # Framework web de alto rendimiento para construir APIs con Python.
uvicorn                     # Servidor ASGI necesario para ejecutar aplicaciones FastAPI.

# --- Base de Datos ---
sqlmodel                    # ORM moderno que combina SQLAlchemy y Pydantic (NECESARIO PARA TU TFG).
# IMPORTANTE: versión acotada a la rama 2.0 a propósito.
# A partir de SQLAlchemy 2.1, una URL "postgresql://" usa por defecto el driver psycopg (v3)
# en lugar de psycopg2. Como la imagen solo instala psycopg2-binary, actualizar sin control
# rompe el arranque de la API (ModuleNotFoundError: No module named 'psycopg').
# Ver docs/INCIDENCIA_LOGIN_SQLALCHEMY_PSYCOPG.md
sqlalchemy>=2.0,<2.1        # ORM (Mapeo Objeto-Relacional) para interactuar con SQL desde Python.
psycopg2-binary             # Adaptador/Driver para conectar Python con PostgreSQL.

# --- Validación de Datos ---
pydantic                    # Librería para validación de datos y definición de esquemas (Schemas).

# --- Seguridad y Autenticación ---
python-multipart            # Soporte para formularios y carga de archivos (multipart/form-data).
python-jose[cryptography]   # Generación y validación de tokens JWT (JSON Web Tokens).
passlib[bcrypt]             # Librería de gestión de contraseñas y hashing.
bcrypt==4.0.1               # Algoritmo de hashing específico para encriptar contraseñas.

# --- Conexiones Externas ---
requests                    # Cliente HTTP para realizar peticiones a APIs externas (Perenual).
